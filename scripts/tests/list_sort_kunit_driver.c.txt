void list_sort(void *ctx,struct list_head *list,list_cmp_func_t cmp) {
    head=list;original_cmp=cmp;
    if(mode>=5 && mode<=8) {
        unsigned serial=elements[0]->serial, poison=elements[0]->poison1;
        if(mode==5 || mode==8) elements[0]->poison1=0;
        if(mode==6 || mode==8) { elements[0]->serial=642;table[642]=elements[0]; }
        if(mode==7) table[0]=elements[1];
        if(mode==8) {
            elements[0]->poison2=0;elements[1]->poison1=0;elements[1]->poison2=0;
            elements[1]->serial=643;table[642]=NULL;table[643]=NULL;
        }
        trace_cmp(ctx,&elements[0]->list,&elements[1]->list);
        elements[0]->serial=serial;elements[0]->poison1=poison;table[0]=elements[0];
        if(mode==8) {
            elements[0]->poison2=0xA324354C;elements[1]->poison1=0xDEADBEEF;
            elements[1]->poison2=0xA324354C;elements[1]->serial=1;
        }
    }
    provider_sort(ctx,list,trace_cmp);
    if(mode==9) list->next->next->prev=list;
    if(mode==10) {
        struct debug_el *a=(void *)((char *)list->next-__builtin_offsetof(struct debug_el,list));
        a->value=214;
    }
    if(mode==11) {
        struct list_head *a=list->next;
        while(a->next!=list) {
            struct debug_el *x=(void *)((char *)a-__builtin_offsetof(struct debug_el,list));
            struct debug_el *y=(void *)((char *)a->next-__builtin_offsetof(struct debug_el,list));
            if(x->value==y->value) {
                unsigned tmp=x->serial;x->serial=y->serial;y->serial=tmp;
                table[x->serial]=x;table[y->serial]=y;break;
            }
            a=a->next;
        }
    }
    if(mode==12) list->prev=list;
    if(mode==13) { list->next=list->next->next;list->next->prev=list; }
}
