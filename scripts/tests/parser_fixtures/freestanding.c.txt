/* Test runtime for ELF32. No kernel syscall result is interpreted as a pass. */
typedef __SIZE_TYPE__ size_t;
static unsigned char arena[16384];
static size_t used;
void *malloc(size_t n)
{
    size_t at = (used + 15) & ~(size_t)15;
    if (n > sizeof(arena) - at) return (void *)0;
    used = at + n;
    return arena + at;
}
void free(void *p) { (void)p; }
void *memcpy(void *dst, const void *src, size_t n)
{
    unsigned char *d = dst; const unsigned char *s = src;
    while (n--) *d++ = *s++;
    return dst;
}
void *memset(void *dst, int ch, size_t n)
{
    unsigned char *d = dst;
    while (n--) *d++ = ch;
    return dst;
}
int memcmp(const void *a, const void *b, size_t n)
{
    const unsigned char *x = a, *y = b;
    while (n--) { if (*x != *y) return *x - *y; ++x; ++y; }
    return 0;
}
size_t strlen(const char *s) { size_t n = 0; while (s[n]) ++n; return n; }
char *strcpy(char *d, const char *s) { char *r = d; while ((*d++ = *s++)) {} return r; }
int strcmp(const char *a, const char *b)
{
    while (*a && *a == *b) { ++a; ++b; }
    return (unsigned char)*a - (unsigned char)*b;
}
int strncmp(const char *a, const char *b, size_t n)
{
    while (n--) {
        if (*a != *b || !*a) return (unsigned char)*a - (unsigned char)*b;
        ++a; ++b;
    }
    return 0;
}
char *strchr(const char *s, int c)
{
    do { if (*s == (char)c) return (char *)s; } while (*s++);
    return (void *)0;
}
static void exit_status(int status) __attribute__((noreturn));
static void exit_status(int status)
{
    __asm__ volatile("int $0x80" : : "a"(1), "b"(status) : "memory");
    __builtin_unreachable();
}
void abort(void) { exit_status(99); }
int printf(const char *s, ...) { (void)s; return 0; }
int puts(const char *s)
{
    size_t n = strlen(s);
    __asm__ volatile("int $0x80" : : "a"(4), "b"(1), "c"(s), "d"(n) : "memory");
    return 0;
}
extern int main(int, char **);
/* Process entry has no return address: establish the i386 C call alignment. */
__attribute__((naked, noreturn)) void _start(void)
{
    __asm__ volatile(
        "andl $-16, %esp\n"
        "subl $8, %esp\n"
        "pushl $0\n"
        "pushl $0\n"
        "call main\n"
        "movl %eax, %ebx\n"
        "movl $1, %eax\n"
        "int $0x80\n"
        "ud2\n");
}
